# DRUMku — a native VST3 MIDI drum sampler for Haiku (no framework).
#
# Builds two ways:
#   - standalone: point VST3_SDK_DIR at a Haiku-patched VST 3 SDK checkout
#     (default: ../VST3-renku/vst3sdk) and configure this directory directly;
#   - as a subdirectory of a project that already builds that SDK (the parent
#     must provide the SDK targets and public_sdk_SOURCE_DIR).

cmake_minimum_required(VERSION 3.25.0)

project(smtg-drumku
    VERSION 0.5.0
    DESCRIPTION "DRUMku MIDI drum sampler VST3 for Haiku"
)

if(NOT TARGET sdk)
    set(VST3_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../VST3-renku/vst3sdk"
        CACHE PATH "Haiku-patched VST 3 SDK source tree")
    if(NOT EXISTS ${VST3_SDK_DIR}/CMakeLists.txt)
        message(FATAL_ERROR
            "VST 3 SDK not found at '${VST3_SDK_DIR}' - set -DVST3_SDK_DIR=<path>")
    endif()

    # Headless SDK: core libraries plus the validator (part of the hosting
    # examples), no VSTGUI (no Haiku backend yet), no plug-in examples.
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "VSTGUI has no Haiku backend yet" FORCE)
    set(SMTG_ENABLE_WAYLAND_SUPPORT OFF CACHE BOOL "no Wayland on Haiku" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "validator" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "plug-in examples not needed" FORCE)
    set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "run manually instead" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "install manually instead" FORCE)

    # The audio backends the standalone app plays on, defined before the SDK
    # (its audiohost sample looks for them too).
    include(${VST3_SDK_DIR}/../haiku/jdbackend.cmake OPTIONAL)

    add_subdirectory(${VST3_SDK_DIR} vst3sdk)

    # The SDK's plugin helper functions read these directory-scoped variables;
    # the SDK sets them only in its own scope.
    set(public_sdk_SOURCE_DIR "${VST3_SDK_DIR}/public.sdk")
    set(pluginterfaces_SOURCE_DIR "${VST3_SDK_DIR}/pluginterfaces")
endif()

# ---------------------------------------------------------------------------
# The plug-in
# ---------------------------------------------------------------------------
smtg_add_vst3plugin(DRUMku
    source/idrumloader.h
    source/drumids.h
    source/version.h
    source/wavsample.h
    source/wavsample.cpp
    source/drumvoices.h
    source/drumvoices.cpp
    source/drumprocessor.h
    source/drumprocessor.cpp
    source/drumcontroller.h
    source/drumcontroller.cpp
    source/drumview.h
    source/drumview.cpp
    source/drumfactory.cpp
)

smtg_target_configure_version_file(DRUMku)

# Editor sprite layers (generated by gui/make_*.sh, committed) -> bundle
# Contents/Resources/gui/.
file(GLOB DRUMKU_GUI_PNGS "${CMAKE_CURRENT_SOURCE_DIR}/resource/gui/*.png")
smtg_target_add_plugin_resources(DRUMku
    RESOURCES ${DRUMKU_GUI_PNGS}
    OUTPUT_SUBDIRECTORY gui
)

target_compile_features(DRUMku
    PUBLIC
        cxx_std_17
)
target_compile_options(DRUMku PRIVATE -ffast-math)

# Native Haiku editor: the HaikuPlugView helper header lives next to the
# Haiku-patched SDK tree (VST3-haiku/haiku), and the view links the Be API.
if(DEFINED VST3_SDK_DIR)
    set(DRUMKU_HAIKU_SUPPORT_DIR "${VST3_SDK_DIR}/../haiku")
else()
    set(DRUMKU_HAIKU_SUPPORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../VST3-renku/haiku")
endif()
if(NOT EXISTS "${DRUMKU_HAIKU_SUPPORT_DIR}/haikuplugview.h")
    message(FATAL_ERROR
        "HaikuPlugView helper not found at '${DRUMKU_HAIKU_SUPPORT_DIR}' - "
        "check out VST3-haiku next to this repository")
endif()
target_include_directories(DRUMku PRIVATE "${DRUMKU_HAIKU_SUPPORT_DIR}")

target_link_libraries(DRUMku
    PRIVATE
        sdk
        be
        tracker # BFilePanel
        translation
)

# ---------------------------------------------------------------------------
# The standalone app: DRUMku without a DAW, played from MIDI sources on the
# Media Kit or on JACK. It hosts the DRUMku.vst3 bundle built above through the
# SDK audiohost's AudioClient on JackDAW's audio backends (VST3-renku's
# haiku/jdbackend): the Media Kit built in, JACK as the jack.so add-on, so it
# never links libjack.
# ---------------------------------------------------------------------------
if(HAIKU)
    include("${DRUMKU_HAIKU_SUPPORT_DIR}/jdbackend.cmake")
    set(DRUMKU_SDK_DIR "${DRUMKU_HAIKU_SUPPORT_DIR}/../vst3sdk")
    set(DRUMKU_AUDIOHOST_DIR "${DRUMKU_SDK_DIR}/public.sdk/samples/vst-hosting/audiohost")

    add_executable(DRUMkuApp
        standalone/main.cpp
        standalone/mainwindow.cpp
        standalone/pluginhost.cpp
        standalone/settings.cpp
        ${DRUMKU_AUDIOHOST_DIR}/source/media/audioclient.cpp
        ${DRUMKU_AUDIOHOST_DIR}/source/media/jdbackend/jdbackendclient.cpp
        ${DRUMKU_SDK_DIR}/public.sdk/source/vst/hosting/plugprovider.cpp
        ${DRUMKU_SDK_DIR}/public.sdk/source/vst/hosting/module_haiku.cpp
        ${DRUMKU_SDK_DIR}/public.sdk/source/common/memorystream.cpp
    )
    target_compile_features(DRUMkuApp PRIVATE cxx_std_17)
    # Warnings for our own files; the SDK's are built as the SDK builds them.
    set_source_files_properties(
        standalone/main.cpp standalone/mainwindow.cpp standalone/pluginhost.cpp
        standalone/settings.cpp
        PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra")
    target_include_directories(DRUMkuApp PRIVATE ${DRUMKU_SDK_DIR} source)
    # tracker: BFilePanel (the preset open/save panels).
    target_link_libraries(DRUMkuApp PRIVATE sdk_hosting jdbackend be tracker)
    set_target_properties(DRUMkuApp
        PROPERTIES
            OUTPUT_NAME DRUMku
            RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/$<CONFIG>
    )
    # The bundle it hosts, and the JACK add-on it can load.
    add_dependencies(DRUMkuApp DRUMku)
    if(TARGET jdbackend_jack)
        add_dependencies(DRUMkuApp jdbackend_jack)
    endif()

    # Signature, launch flags, version and icon, so Tracker and Deskbar treat
    # it as an application.
    set(DRUMKU_RSRC ${CMAKE_CURRENT_BINARY_DIR}/DRUMku.rsrc)
    add_custom_command(OUTPUT ${DRUMKU_RSRC}
        COMMAND rc -I ${CMAKE_CURRENT_SOURCE_DIR}/standalone -o ${DRUMKU_RSRC}
                ${CMAKE_CURRENT_SOURCE_DIR}/standalone/DRUMku.rdef
        DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/standalone/DRUMku.rdef
                ${CMAKE_CURRENT_SOURCE_DIR}/standalone/icons/drumku_icon.rdef
        COMMENT "Compiling the DRUMku app resources"
    )
    add_custom_target(DRUMkuAppResources DEPENDS ${DRUMKU_RSRC})
    add_dependencies(DRUMkuApp DRUMkuAppResources)
    add_custom_command(TARGET DRUMkuApp POST_BUILD
        COMMAND xres -o $<TARGET_FILE:DRUMkuApp> ${DRUMKU_RSRC}
        COMMAND mimeset -f $<TARGET_FILE:DRUMkuApp>
    )
endif()
